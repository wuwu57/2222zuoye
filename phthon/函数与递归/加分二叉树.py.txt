
def pre_order(i, j):
    if i > j:
        return
    k = root[i][j]
    ans.append(str(k))
    pre_order(i, k - 1)
    pre_order(k + 1, j)

n = int(input())
score = list(map(int, input().split()))
score = [0] + score
dp = [[0] * (n + 2) for _ in range(n + 2)]
root = [[0] * (n + 2) for _ in range(n + 2)]

for i in range(1, n + 1):
    dp[i][i] = score[i]
    root[i][i] = i

for length in range(2, n + 1):
    for i in range(1, n - length + 2):
        j = i + length - 1
        for k in range(i, j + 1):
            left = dp[i][k - 1] if i <= k - 1 else 1
            right = dp[k + 1][j] if k + 1 <= j else 1
            val = left * right + score[k]
            if val > dp[i][j]:
                dp[i][j] = val
                root[i][j] = k

ans = []
pre_order(1, n)
print(dp[1][n])
print(' '.join(ans))
