import sys

def main():
    data = list(map(int, sys.stdin.read().split()))
    ptr = 0
    n = data[ptr]
    m = data[ptr+1]
    ptr += 2
    
    pre = [0]*(n+1)
    for i in range(1, n+1):
        pre[i] = pre[i-1] + data[ptr]
        ptr += 1
    
    res = []
    for _ in range(m):
        L = data[ptr]
        R = data[ptr+1]
        ptr += 2
        res.append(str(pre[R] - pre[L-1]))
    print('\n'.join(res))

if __name__ == "__main__":
    main()
