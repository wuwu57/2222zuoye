res = []
for a in range(1, 100):
    for b in range(a + 1, 10000):
        c = a * b
        s = str(a) + str(b) + str(c)
        if len(s) == 9 and len(set(s)) == 9 and '0' not in s:
            res.append((c, a, b))
res.sort(key=lambda x: (x[0], x[1]))
for item in res:
    print(f"{item[0]} = {item[1]} x {item[2]}")
